function callSheetPulse(text, inst, action) {
  if (!text) return "";
  if (!SHEETPULSE_API_KEY || SHEETPULSE_API_KEY === "YOUR_API_KEY_HERE") {
    return "Error: Please generate or activate your SheetPulse API key.";
  }
  const payload = {
    text: String(text),
    instruction: String(inst || ""),
    action: action,
    api_key: SHEETPULSE_API_KEY
  };
  const res = UrlFetchApp.fetch(BACKEND_URL, {
    method: "post",
    contentType: "application/json",
    payload: JSON.stringify(payload),
    muteHttpExceptions: true
  });
  const json = JSON.parse(res.getContentText());
  return json.success ? json.result : "Error: " + (json.detail || "API Error");
}
